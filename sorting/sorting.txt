Insertion Sort 

#include <vector>
using namespace std;

void insertionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j = j - 1;
        }
        arr[j + 1] = key;
    }
}



Maan lo array hai: `[4, 2, 1, 3]`

* **Initial State:** `[4, 2, 1, 3]` (Sirf pehla element `4` sorted maana jata hai)
* **Step 1 (i = 1, key = 2):**
* `key` (2) ko pichle element (4) se compare kiya.
* 4 > 2, isliye 4 ko ek jagah aage khiskaya.
* 2 ko sahi jagah (index 0) par daala.
* **Current Array:** `[2, 4, 1, 3]`


* **Step 2 (i = 2, key = 1):**
* `key` (1) ko pichle elements (4 aur 2) se compare kiya.
* 4 > 1, 4 ko aage khiskaya.
* 2 > 1, 2 ko aage khiskaya.
* 1 ko sahi jagah (index 0) par daala.
* **Current Array:** `[1, 2, 4, 3]`


* **Step 3 (i = 3, key = 3):**
* `key` (3) ko pichle elements se compare kiya.
* 4 > 3, 4 ko aage khiskaya.
* 2 < 3, shifting rok di.
* 3 ko 2 ke aage (index 2) par daala.
* **Final Array:** `[1, 2, 3, 4]`

















Selection Sort

#include <iostream>
using namespace std;

void selectionSort(int arr[], int n)
{
    for(int i = 0; i < n - 1; i++)
    {
        int minIndex = i;

        for(int j = i + 1; j < n; j++)
        {
            if(arr[j] < arr[minIndex])
            {
                minIndex = j;
            }
        }

        swap(arr[i], arr[minIndex]);
    }
}

int main()
{
    int arr[] = {4, 2, 5, 1, 3};
    int n = sizeof(arr) / sizeof(arr[0]);

    selectionSort(arr, n);

    for(int i = 0; i < n; i++)
    {
        cout << arr[i] << " ";
    }

    return 0;
}

# Merge Sort

## C++ Code

```cpp
#include <iostream>
#include <vector>
using namespace std;

void merge(vector<int>& arr, int low, int mid, int high)
{
    vector<int> temp;

    int left = low;
    int right = mid + 1;

    while(left <= mid && right <= high)
    {
        if(arr[left] <= arr[right])
        {
            temp.push_back(arr[left]);
            left++;
        }
        else
        {
            temp.push_back(arr[right]);
            right++;
        }
    }

    while(left <= mid)
    {
        temp.push_back(arr[left]);
        left++;
    }

    while(right <= high)
    {
        temp.push_back(arr[right]);
        right++;
    }

    for(int i = low; i <= high; i++)
    {
        arr[i] = temp[i - low];
    }
}

void mergeSort(vector<int>& arr, int low, int high)
{
    if(low >= high)
        return;

    int mid = (low + high) / 2;

    mergeSort(arr, low, mid);
    mergeSort(arr, mid + 1, high);

    merge(arr, low, mid, high);
}

int main()
{
    vector<int> arr = {4, 2, 5, 1, 3};

    mergeSort(arr, 0, arr.size() - 1);

    for(int x : arr)
        cout << x << " ";

    return 0;
}
```

### Final Merge

The merge function is called like this:

```cpp
merge(arr, low, mid, high);
```

For the final merge, the array is split into:

```
[2,4,5]

and

[1,3]
```

So the actual function call becomes:

```cpp
merge(arr, 0, 2, 4);
```

Here:

* low = 0
* mid = 2
* high = 4

Inside the merge function, the temp array is filled first by comparing elements from the left and right halves.

Compare one by one:

```
2 vs 1 → 1

2 vs 3 → 2

4 vs 3 → 3

Remaining → 4 5
```

Now temp contains:

```cpp
[1,2,3,4,5]
```

After that, this for loop copies the values from temp back into the original array:

```cpp
for(int i = low; i <= high; i++)
{
    arr[i] = temp[i - low];
}
```

How this loop works:

* i starts from low and goes up to high
* temp[i - low] is used because temp starts from index 0
* when i = low, arr[low] gets temp[0]
* when i = low + 1, arr[low + 1] gets temp[1]
* this continues until all values are copied back

So after the loop finishes, the final sorted result is:

```cpp
[1,2,3,4,5]

# Easy Trick to Remember

Merge Sort always follows **two steps**:

### 1. Divide

Keep dividing the array into two halves until every subarray has only one element.

### 2. Merge

Merge the smaller sorted arrays back together by comparing elements one by one.

**Memory Formula:**

**Divide → Divide → Divide → Merge → Merge → Merge**

Quick Sort
C++ Code
#include <iostream>
#include <vector>
using namespace std;

int partition(vector<int>& arr, int low, int high)
{
    int pivot = arr[low];

    int i = low;
    int j = high;

    while(i < j)
    {
        while(arr[i] <= pivot && i < high)
            i++;

        while(arr[j] > pivot && j > low)
            j--;

        if(i < j)
            swap(arr[i], arr[j]);
    }

    swap(arr[low], arr[j]);

    return j;
}

void quickSort(vector<int>& arr, int low, int high)
{
    if(low < high)
    {
        int pivotIndex = partition(arr, low, high);

        quickSort(arr, low, pivotIndex - 1);
        quickSort(arr, pivotIndex + 1, high);
    }
}

int main()
{
    vector<int> arr = {4, 2, 5, 1, 3};

    quickSort(arr, 0, arr.size() - 1);

    for(int x : arr)
        cout << x << " ";

    return 0;
}
